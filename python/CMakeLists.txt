cmake_minimum_required(VERSION 3.21)
project(zspace_python LANGUAGES CXX)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PYBIND11_FINDPYTHON ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
set(ZSPACE_CORE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../zspace_core" CACHE PATH "Core source tree")
set(ZSPACE_TOOLSETS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "v3 toolsets source tree")
set(ZSPACE_BUILD_INTERFACE ON CACHE BOOL "" FORCE)
set(ZSPACE_BUILD_IO ON CACHE BOOL "" FORCE)
set(ZSPACE_IO_WITH_OPENUSD OFF CACHE BOOL "" FORCE)
set(ZSPACE_IO_WITH_TINYUSDZ OFF CACHE BOOL "" FORCE)
foreach(opt DISPLAY INTEROP TESTS)
  set(ZSPACE_BUILD_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(ZSPACE_WITH_OPENGL OFF CACHE BOOL "" FORCE)
set(ZSPACE_BUILD_SHARED ON CACHE BOOL "" FORCE)
add_subdirectory("${ZSPACE_CORE_SOURCE}" "${CMAKE_BINARY_DIR}/native-core")
set(ZSPACE_PYTHON_PACKAGE_DIR "${CMAKE_BINARY_DIR}/package/zspace")
file(MAKE_DIRECTORY "${ZSPACE_PYTHON_PACKAGE_DIR}/core" "${ZSPACE_PYTHON_PACKAGE_DIR}/toolset/slicer")
file(WRITE "${ZSPACE_PYTHON_PACKAGE_DIR}/__init__.py" "from . import core\n")
file(WRITE "${ZSPACE_PYTHON_PACKAGE_DIR}/core/__init__.py" "import os, sys\nif sys.platform == 'win32':\n    _dll_dir = os.add_dll_directory(os.path.dirname(__file__))\nfrom ._core import *\n")
file(WRITE "${ZSPACE_PYTHON_PACKAGE_DIR}/toolset/__init__.py" "")
file(WRITE "${ZSPACE_PYTHON_PACKAGE_DIR}/toolset/slicer/__init__.py" "import os, sys\nfrom zspace import core\nif sys.platform == 'win32':\n    _dll_dir = os.add_dll_directory(os.path.dirname(__file__))\nfrom ._slicer import *\n")
add_subdirectory(core)
option(ZSPACE_PYTHON_BUILD_SLICER "Build the slicer extension" ON)
option(ZSPACE_PYTHON_WITH_USD "Enable USD IO in slicer bindings" OFF)
if(ZSPACE_PYTHON_BUILD_SLICER)
  set(ZSPACE_TOOLSETS_SLICING_ONLY ON CACHE BOOL "" FORCE)
  set(ZSPACE_TOOLSETS_BUILD_SHARED ON CACHE BOOL "" FORCE)
  set(ZSPACE_TOOLSETS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSPACE_CORE_DIR "${ZSPACE_CORE_SOURCE}" CACHE PATH "" FORCE)
  set(ZSPACE_TOOLSETS_SLICING_WITH_OPENUSD ${ZSPACE_PYTHON_WITH_USD} CACHE BOOL "" FORCE)
  add_subdirectory("${ZSPACE_TOOLSETS_SOURCE}" "${CMAKE_BINARY_DIR}/native-toolsets")
  add_subdirectory(toolsets/zTs3DP)
endif()
