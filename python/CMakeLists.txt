cmake_minimum_required(VERSION 3.21)
project(zspace_python LANGUAGES CXX)
set(ZSPACE_CORE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../zspace_core" CACHE PATH "Core source tree")
set(ZSPACE_TOOLSETS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "v3 toolsets source tree")
include("${ZSPACE_CORE_SOURCE}/python/HeadlessPython.cmake")
add_subdirectory("${ZSPACE_CORE_SOURCE}" "${CMAKE_BINARY_DIR}/native-core")
set(ZSPACE_PYTHON_PACKAGE_DIR "${CMAKE_BINARY_DIR}/package/zspace" CACHE PATH "Python package output")
file(MAKE_DIRECTORY "${ZSPACE_PYTHON_PACKAGE_DIR}/core" "${ZSPACE_PYTHON_PACKAGE_DIR}/toolset/slicer")
file(WRITE "${ZSPACE_PYTHON_PACKAGE_DIR}/toolset/__init__.py" "")
file(WRITE "${ZSPACE_PYTHON_PACKAGE_DIR}/toolset/slicer/__init__.py" "from zspace import core\nfrom ._slicer import *\n")
add_subdirectory("${ZSPACE_CORE_SOURCE}/python/core" "${CMAKE_BINARY_DIR}/python-core")
option(ZSPACE_PYTHON_BUILD_SLICER "Build the slicer extension" ON)
if(ZSPACE_PYTHON_BUILD_SLICER)
  set(ZSPACE_TOOLSETS_WITH_USD_IO ${ZSPACE_PYTHON_WITH_USD} CACHE BOOL "Slicer IO via Core" FORCE)
  set(ZSPACE_TOOLSETS_SLICING_ONLY ON CACHE BOOL "" FORCE)
  set(ZSPACE_TOOLSETS_BUILD_SHARED ON CACHE BOOL "" FORCE)
  set(ZSPACE_TOOLSETS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSPACE_CORE_DIR "${ZSPACE_CORE_SOURCE}" CACHE PATH "" FORCE)
  add_subdirectory("${ZSPACE_TOOLSETS_SOURCE}" "${CMAKE_BINARY_DIR}/native-toolsets")
  add_subdirectory(toolsets/zTs3DP)
endif()

if(ZSPACE_PYTHON_WITH_USD)
file(MAKE_DIRECTORY "${ZSPACE_PYTHON_PACKAGE_DIR}/io")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/adapters/slicing_usd.py" "${ZSPACE_PYTHON_PACKAGE_DIR}/io/slicing_usd.py" COPYONLY)

endif()
